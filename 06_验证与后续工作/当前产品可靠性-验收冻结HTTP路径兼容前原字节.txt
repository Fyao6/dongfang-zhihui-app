def sha(file):
    digest = hashlib.sha256()
    with Path(file).open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def read(file):
    def pairs(values):
        result = {}
        for key, value in values:
            assert key not in result, 'duplicate JSON key'
            result[key] = value
        return result
    return json.loads(Path(file).read_text('utf-8'), object_pairs_hook=pairs,
                      parse_constant=lambda _: (_ for _ in ()).throw(AssertionError('nonfinite JSON')))


def project_file(name):
    assert isinstance(name, str) and not Path(name).is_absolute(), 'relative project evidence required'
    name = name.replace('\\', '/')
    assert '..' not in name.split('/'), 'canonical relative project path required'
    result = (ROOT / name).resolve()
    assert result.is_relative_to(ROOT) and result.is_file(), 'project evidence missing or outside root'
    return result


def verify_repair(entry):
    """Only the shared current-product proof permits actual operation evidence."""
    assert isinstance(entry, dict) and entry.get('path') == REPAIR.relative_to(ROOT).as_posix()
    digest = entry.get('sha256')
    assert isinstance(digest, str) and HEX.fullmatch(digest) and sha(REPAIR) == digest
    authority = {'path': REPAIR_AUTHORIZATION.relative_to(ROOT).as_posix(),
                 'sha256': REPAIR_AUTHORIZATION_SHA256}
    assert entry.get('authorization') == authority and sha(REPAIR_AUTHORIZATION) == REPAIR_AUTHORIZATION_SHA256
    environment = {key: os.environ[key] for key in ('SystemRoot', 'WINDIR', 'TEMP', 'TMP', 'COMSPEC') if key in os.environ}
    environment['PATH'] = str(NODE.parent)
    result = subprocess.run([str(NODE), str(REPAIR_VERIFIER), '--verify-repair-proof',
        '--repair-proof-sha256=' + digest, '--repair-authorization-sha256=' + REPAIR_AUTHORIZATION_SHA256],
        cwd=ROOT, env=environment, shell=False, capture_output=True, timeout=90,
        creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
    assert result.returncode == 0 and len(result.stdout) <= 262144, 'shared current-product proof verification failed'
    checked = json.loads(result.stdout.decode('utf-8'))
    assert checked.get('pass') is True and checked.get('stage') == 31 and checked.get('startsService') is False
    assert checked.get('repairProof') == {'path': entry['path'], 'sha256': digest}
    assert checked.get('authorization') == dict(authority, revision=4)
    assert checked['stage30ProofAnchor']['path'] == PROOF.relative_to(ROOT).as_posix()
    assert checked['stage30ProofAnchor']['sha256'] == PROOF_SHA256
    assert checked['stage30ProofAnchor']['originalStage30ProductUnchanged'] is False
    repair, authorization = read(REPAIR), read(REPAIR_AUTHORIZATION)
    assert repair['kind'] == 'current-product-reliability-evidence-binding' and repair['opsExecutionAllowed']
    assert authorization['revision'] == 4
    current = authorization['currentProductHashes']
    assert len(current) == authorization['currentProductInputCount'] and checked['currentSourceAndBuildHashes'] == current
    for name, expected in repair['reportIdentityHashes'].items():
        assert HEX.fullmatch(expected) and sha(project_file(name)) == expected, 'current execution input changed: ' + name
    assert all(repair['reportIdentityHashes'][name] == digest for name, digest in current.items())
    return repair, authorization, {
        'path': entry['path'], 'sha256': digest, 'authorization': authority,
        'sharedVerifier': {'path': REPAIR_VERIFIER.relative_to(ROOT).as_posix(), 'sha256': sha(REPAIR_VERIFIER)},
        'sharedActualRepairVerifierUsed': True, 'candidateAloneAllowsExecution': False,
        'perBackendPassed': checked['perBackendPassed'], 'inputCount': len(repair['reportIdentityHashes']),
        'currentProductInputCount': authorization['currentProductInputCount'], 'stage30ProofAnchor': checked['stage30ProofAnchor'],
        'originalStage30ProductUnchanged': False, 'revision': 4,
        'historicalInputSets': authorization['historicalInputSets'],
        'currentStableBuild': authorization['currentStableBuild'],
        'authorizedProductChanges': authorization['authorizedProductChanges']}


def verify_local_artifacts(entries):
    """Verify private files or a complete owned directory manifest without exporting them."""
    assert isinstance(entries, list)
    verified, roots, provenance_cache = [], [], {}
    for entry in entries:
        assert isinstance(entry, dict) and entry.get('includedInDefaultHandoff') is False
        name = entry.get('path')
        assert isinstance(name, str) and not Path(name).is_absolute() and '\\' not in name and '..' not in name.split('/')
        file = ROOT / name
        assert file.exists() and file.resolve().is_relative_to(ROOT)
        for current in (file, *file.parents):
            if current == ROOT.parent:
                break
            info = current.lstat()
            assert not stat.S_ISLNK(info.st_mode) and not (getattr(info, 'st_file_attributes', 0) & 0x400), 'reparse/link artifacts refused'
        file = file.resolve()
        if file.parent == OUT:
            assert file.name.startswith(PREFIX), 'owned local-only file required'
        else:
            managed = ROOT / '04_研发源码/app/cloud/.wrangler'
            assert file.is_relative_to(managed)
            owned = file.relative_to(managed).parts[0]
            anchor = entry.get('provenance')
            assert isinstance(anchor, dict) and HEX.fullmatch(str(anchor.get('sha256', ''))), 'owned-state provenance anchor required'
            authority = project_file(anchor['path'])
            assert authority.parent == OUT and authority.name.startswith(PREFIX)
            assert sha(authority) == anchor['sha256']
            key = (str(authority), anchor['sha256'])
            if key not in provenance_cache:
                provenance_cache[key] = read(authority)
            origin = provenance_cache[key]
            if isinstance(origin.get('core'), dict):
                core = origin['core']
                run = core.get('runId')
                assert re.fullmatch('[a-f0-9]{32}', str(run)) and core.get('backend') == 'cloud'
                assert core.get('status') == 'armed-local-only' and core.get('productionEnabled') is False
                namespace = core.get('cloudStateBasename', 'local-reliability-recovery-' + run)
                assert namespace in ('rl-' + run, 'local-reliability-recovery-' + run)
                assert owned in (namespace, namespace + '-restored', namespace + '-rollback')
            else:
                run = origin.get('runId')
                assert re.fullmatch('[a-f0-9]{32}', str(run))
                declared = origin.get('ownedState') or origin.get('ownedFreshState')
                assert isinstance(declared, str)
                declared = (ROOT / declared).resolve()
                assert declared == managed / owned, 'selected evidence must name this exact state'
                if origin.get('role') == 'mediaCloud':
                    expected_name = PREFIX + '后端媒体隔离-cloud-' + run
                elif origin.get('mode') in ('fixed-30-minute-load', 'interface-component-only'):
                    assert origin.get('target') == 'cloud'
                    expected_name = PREFIX + '隔离-cloud-' + run
                elif origin.get('kind') == 'strict-old-snapshot-new-native-import-bridge':
                    assert owned in ('lrni-' + run, 'local-reliability-native-import-debug-' + run)
                    expected_name = owned
                elif origin.get('kind') == 'actual-native-d1-import-diagnosis':
                    expected_name = 'local-reliability-import-debug-' + run
                else:
                    raise AssertionError('unknown private-state provenance schema')
                assert owned == expected_name, 'explicit owned local state required'
        assert all(not (file == old or file.is_relative_to(old) or old.is_relative_to(file)) for old in roots), 'overlapping private artifact declarations'
        roots.append(file)
        if file.is_file():
            assert entry.get('kind', 'file') == 'file' and HEX.fullmatch(str(entry.get('sha256', '')))
            assert sha(file) == entry['sha256'], 'private file changed'
            verified.append(dict(entry, bytes=file.stat().st_size))
            continue
        assert file.is_dir() and entry.get('kind') == 'directory'
        manifest_entry = entry.get('manifest')
        assert isinstance(manifest_entry, dict) and HEX.fullmatch(str(manifest_entry.get('sha256', '')))
        manifest_path = project_file(manifest_entry['path'])
        assert manifest_path.parent == OUT and manifest_path.name.startswith(PREFIX)
        assert sha(manifest_path) == manifest_entry['sha256'], 'private directory manifest changed'
        manifest = read(manifest_path)
        assert manifest.get('schemaVersion') == 1 and manifest.get('root') == name
        expected = manifest.get('files')
        assert isinstance(expected, dict) and expected, 'complete nonempty private directory manifest required'
        actual = {}
        for directory, subdirs, files in os.walk(file, followlinks=False):
            for child in subdirs + files:
                item = Path(directory) / child
                info = item.lstat()
                assert not stat.S_ISLNK(info.st_mode) and not (getattr(info, 'st_file_attributes', 0) & 0x400)
                assert item.name != '.dev.vars', 'private environment file must already be removed'
            for child in files:
                item = Path(directory) / child
                assert item.is_file()
                actual[item.relative_to(file).as_posix()] = sha(item)
        assert actual == expected, 'private directory changed or manifest incomplete'
        verified.append(dict(entry, fileCount=len(actual), manifestVerified=True))
    return verified, roots


def collect():
    """Verify selected completed evidence, returning deterministic fields except time.

    The selection file is an explicit root-authored allowlist, never a prefix glob.
    Its role entries contain path, sha256, childExitCode; aliases are explicit
    (originalPath, sha256, path) tuples. Diagnostic/tool/guide lists are explicit.
    """
    assert sha(BASELINE) == BASELINE_SHA256
    baseline = read(BASELINE)
    assert baseline['pass'] and baseline['nextLocalStage'] == 31
    assert baseline['productChangesAuthorizedForThisExercise'] is False
    assert sha(PROOF) == PROOF_SHA256
    proof = read(PROOF)
    assert proof['pass'] and proof['stage'] == 30
    assert proof['perBackendPassed'] == {'Python': 195, 'Cloud': 195}
    assert proof['sourceAndBuildInputCount'] == 516
    selected = read(SELECTION)
    assert selected['schemaVersion'] == 1 and selected['stage'] == 31
    assert selected['authorizedByRoot'] is True
    assert selected['previousFinalProof'] == {'path': PROOF.relative_to(ROOT).as_posix(), 'sha256': PROOF_SHA256}
    assert set(selected['roles']) == set(ROLES), 'exact eight completed backend roles required'
    repair, authorization, performance_repair = verify_repair(selected['performanceRepair'])
    sources = repair['reportIdentityHashes']
    original_protected = baseline['protectedHistoricalSHA256']
    assert len(original_protected) == 3418
    protected, historical_remaps = {}, {}
    tuple_aliases = {(a['originalPath'], a['sha256']): a['path'] for a in authorization['historicalAliases']}
    assert len(tuple_aliases) == len(authorization['historicalAliases'])
    for name, digest in original_protected.items():
        current = ROOT / name
        if current.is_file() and sha(current) == digest:
            actual_name = name
        else:
            actual_name = tuple_aliases[(name, digest)]
            historical_remaps[name] = {'path': actual_name, 'sha256': digest}
        assert sha(project_file(actual_name)) == digest, 'protected historical bytes changed: ' + name
        assert actual_name not in protected, 'historical preservation path collision'
        protected[actual_name] = digest
    assert len(protected) == 3418 and SERVER in historical_remaps
    receipt = read(OUT / '负责人优化建议-卡组迁移-实施记录.json')
    inherited = receipt['reportIdentityHashes']
    assert len(inherited) == 516
    for name, digest in inherited.items():
        assert proof['verifiedSHA256'][name] == digest
        current = ROOT / name
        historical = name if current.is_file() and sha(current) == digest else tuple_aliases[(name, digest)]
        assert sha(project_file(historical)) == digest
        if current.is_file():
            assert sources[name] == sha(current), 'actual current inherited input missing'
        else:
            assert name not in sources and (name, digest) in tuple_aliases
    evidence = {}
    private_artifacts, private_roots = verify_local_artifacts(selected.get('localOnlyArtifacts', []))
    aliases = dict(tuple_aliases)
    alias_entries = selected.get('historicalInputAliases', [])
    assert isinstance(alias_entries, list), 'aliases are a list keyed by (originalPath, sha256), not one copy per path'
    for alias in alias_entries:
        assert isinstance(alias, dict) and set(alias) == {'originalPath', 'path', 'sha256'}
        name, digest = alias['originalPath'].replace('\\', '/'), alias['sha256']
        assert name.startswith(('06_验证与后续工作/' + PREFIX, '06_验证与后续工作/负责人优化建议-本地可靠性')) and HEX.fullmatch(digest)
        key = (name, digest)
        assert key not in aliases or aliases[key] == alias['path'], 'conflicting historical path/hash alias'
        file = project_file(alias['path'])
        assert file.parent == OUT and file.name.startswith(('负责人优化建议-本地可靠性', PREFIX)) and sha(file) == digest
        aliases[key] = alias['path'].replace('\\', '/')
    private_bindings, runtime_tools, external_tools = {}, {}, {}

    def bind(name, digest, *, historical=True):
        name = name.replace('\\', '/')
        assert HEX.fullmatch(digest), 'invalid SHA256'
        actual_name = name
        candidate = ROOT / name
        if not candidate.is_file() or sha(candidate) != digest:
            assert historical, 'current evidence changed'
            alias = aliases.get((name, digest))
            assert alias, 'preserved original source required for exact path and SHA256'
            actual_name = alias
        file = project_file(actual_name)
        assert sha(file) == digest, 'preserved/current bytes changed'
        if any(file == root or file.is_relative_to(root) for root in private_roots):
            assert actual_name not in private_bindings or private_bindings[actual_name] == digest
            private_bindings[actual_name] = digest
            return file
        if 'node_modules' in file.parts:
            assert actual_name not in runtime_tools or runtime_tools[actual_name] == digest
            runtime_tools[actual_name] = digest
            return file
        assert actual_name not in evidence or evidence[actual_name] == digest
        evidence[actual_name] = digest
        return file

    def item(entry):
        assert isinstance(entry, dict) and {'path', 'sha256'} <= set(entry)
        return bind(entry['path'], entry['sha256'], historical=False)

    def artifact_graph(value):
        if isinstance(value, list):
            for child in value:
                artifact_graph(child)
        elif isinstance(value, dict):
            for path_key, hash_key in [('path', 'sha256'), ('bundle', 'bundleSha256'),
                                       ('metafile', 'metafileSha256')]:
                if isinstance(value.get(path_key), str) and HEX.fullmatch(str(value.get(hash_key, ''))):
                    name = value[path_key]
                    if not Path(name).is_absolute():
                        bind(name, value[hash_key])
            for key in ('inputHashes', 'sourceAndBuildHashes', 'sourceAndToolHashes', 'runtimeToolHashes',
                        'runtimeDependencyHashes', 'reportIdentityHashes', 'sourceCopiesSHA256',
                        'buildPreparationInputHashes', 'currentInheritedInputHashes'):
                mapping = value.get(key)
                if isinstance(mapping, dict):
                    for name, digest in mapping.items():
                        bind(name, digest)
            for child in value.values():
                artifact_graph(child)

    # The shared verifier above checks the real 14-gate + native-safety acceptance.
    # Bind those actual documents and native compilation records as delivery proof,
    # while private DB/state and third-party tool bytes remain separately classified.
    artifact_graph(performance_repair)
    artifact_graph(repair)
    artifact_graph(authorization)
    # Current proof is a union of actual module receipts, not a synthetic fourteen-gate object.
    module_refs = [repair[key] for key in ('growthReceipt', 'websiteReceipt', 'dailyReceipt', 'dailyFullReceipt')]
    module_refs += repair['dailySpecial'] + [repair['requiredCl40']]
    for module_ref in module_refs:
        artifact_graph(read(project_file(module_ref['path'])))
    growth_receipt = read(project_file(repair['growthReceipt']['path']))
    website_receipt = read(project_file(repair['websiteReceipt']['path']))
    daily_full_receipt = read(project_file(repair['dailyFullReceipt']['path']))
    child_refs = growth_receipt['gates'] + list(website_receipt['full'].values())
    child_refs += list(website_receipt['special'].values()) + list(daily_full_receipt['reports'].values())
    child_refs += [entry['actualRouteLauncher'] for entry in website_receipt['special'].values()]
    for child_ref in child_refs:
        artifact_graph(read(project_file(child_ref['path'])))
    previous_repair = read(project_file(authorization['pythonRepairProof']['path']))
    artifact_graph(previous_repair)
    for gate_entry in previous_repair['verification'].values():
        artifact_graph(read(project_file(gate_entry['path'])))

    def common(report):
        assert report.get('productionEnabled') is False
        assert report.get('formalAcceptance') is False
        assert report.get('stage30Proof', {}).get('sha256') == PROOF_SHA256
        if 'performanceRepair' in report:
            binding = report['performanceRepair']
            assert binding['repairProofSha256'] == performance_repair['sha256']
            assert binding['authorizationSha256'] == REPAIR_AUTHORIZATION_SHA256
            assert binding.get('sharedReadOnlyVerifierRequired') is True
            assert binding.get('candidateAloneAllowsExecution') is False
        else:
            assert report.get('repairProof', {}).get('path') == performance_repair['path']
            assert report.get('repairProof', {}).get('sha256') == performance_repair['sha256']
            assert report.get('repairAuthorization', {}).get('path') == performance_repair['authorization']['path']
            assert report.get('repairAuthorization', {}).get('sha256') == REPAIR_AUTHORIZATION_SHA256

    def load(report, target):
        common(report)
        assert report['target'] == target and report['mode'] == 'fixed-30-minute-load'
        assert report['localFunctionalObservationPassed'] is True
        assert report['localBaselineCompleted'] is True and report['thirtyMinuteCompleted'] is True
        assert report['sourceIdentityUnchanged'] is True
        assert report['uniqueAuthenticatedSessions'] == 100
        limits = report['limits']
        for key, expected in [('users', 100), ('rps', 20), ('seconds', 1800), ('requests', 36000)]:
            assert limits[key] == expected, 'short run cannot satisfy full role'
        users = report['users']
        assert len(users) == 100 and len({u['accountId'] for u in users}) == 100
        assert all(u['role'] == 'user' and u['authStatusVerified'] for u in users)
        assert report['measurement']['elapsedSeconds'] >= 1800
        assert report['measurement']['outsideWindowDispatches'] == 0
        raw = report['rawEvidence']
        raw_path = bind(raw['path'], raw['sha256'], historical=False)
        assert raw['complete'] and raw['integrityMatched'] and raw['sha256'] == raw['expectedRawSha256']
        rows, counts, latencies = set(), Counter(), []
        first_dispatch, last_dispatch = math.inf, -math.inf
        with raw_path.open(encoding='utf-8') as stream:
            for line in stream:
                row = json.loads(line)
                assert row['index'] not in rows and 0 <= row['index'] < 36000
                rows.add(row['index'])
                assert row['scheduledMs'] == row['index'] * 50
                assert 0 <= row['dispatchMs'] < 1800000
                assert type(row['dispatchMs']) in (int, float) and math.isfinite(row['dispatchMs'])
                first_dispatch = min(first_dispatch, row['dispatchMs'])
                last_dispatch = max(last_dispatch, row['dispatchMs'])
                assert row['userIndex'] == (row['index'] + row['index'] // 100) % 100
                assert row['status'] == 200 and row['success'] is True and row['error'] is None
                assert math.isfinite(row['latencyMs']) and row['latencyMs'] >= 0
                assert not ({'body', 'cookies', 'Cookie', 'Authorization', 'password', 'sessionHash'} & set(row))
                latencies.append(row['latencyMs'])
                counts[row['method'] + ' ' + row['route']] += 1
        assert len(rows) == raw['actualLineCount'] == raw['expectedCount'] == 36000
        assert counts == Counter({'GET /api/auth/status': 7200, 'GET /api/records': 7200,
            'GET /api/privacy/preferences': 7200, 'GET /api/commerce/cart': 7200,
            'GET /api/cards/draws?environment=sandbox': 6840, 'PATCH /api/privacy/preferences': 360})
        latencies.sort()
        observed = report['measurement']['overall']
        assert observed['failed'] == 0
        for label, percentile in [('p50Ms', .5), ('p95Ms', .95), ('p99Ms', .99)]:
            expected = latencies[math.ceil(len(latencies) * percentile) - 1]
            assert abs(observed[label] - expected) < 0.000001
        writes = report['writeReconciliation']
        assert len(writes) == 100 and sum(row['successfulWrites'] for row in writes) == 360
        assert all(row['matched'] and row['allConsentsDisabled'] and row['unknownOutcomeWrites'] == 0 for row in writes)
        cleanup = report['cleanup']
        assert cleanup['pidInventoryComplete'] and not cleanup['ownedServerStillAlive']
        assert not cleanup['trackedRemainingPids'] and cleanup['privateVarsRemaining'] == 0
        return {'requests': 36000, 'users': 100, 'scheduledRps': 20, 'seconds': 1800,
                'actualElapsedSeconds': report['measurement']['elapsedSeconds'],
                'measurementStartedAt': report['measurementStartedAt'],
                'firstActualDispatchMs': first_dispatch, 'lastActualDispatchMs': last_dispatch,
                'p50Ms': observed['p50Ms'], 'p95Ms': observed['p95Ms'], 'p99Ms': observed['p99Ms'],
                'professional800msThresholdConfirmed': False}

    def recovery(report, target):
        common(report)
        assert report['target'] == target and report['componentOnly'] is False
        assert report['localDrillPassed'] is True and report['status'] == 'local-drill-completed'
        assert report['failed'] == 0 and report['passed'] == len(report['checks']) == 11
        assert all(c['pass'] for c in report['checks'])
        manifest = read(item(report['labManifest']))
        core = manifest['core']
        assert core['runId'] == report['runId'] and core['backend'] == target
        assert core['status'] == 'armed-local-only' and core['explicitExecutionAuthorization'] is True
        assert core['performanceRepair'] == report['performanceRepair']
        assert core['sourceCopiesSHA256'] == report['sourceCopiesSHA256']
        assert hashlib.sha256(json.dumps(core, ensure_ascii=False, sort_keys=True,
            separators=(',', ':')).encode('utf-8')).hexdigest() == manifest['manifestSha256']
        artifact_graph(core)
        names = {c['name'] for c in report['checks']}
        assert {'actual-committed-response-loss-and-idempotent-replay',
                'actual-owned-process-fault-and-same-state-restart',
                'fresh-snapshot-restore-before-any-app-initialization',
                'actual-restored-service-and-nonzero-rpo-reconciliation',
                'candidate-write-and-armed-lab-rollback-to-new-target'} <= names
        assert report['cleanup']['allTrackedPidsExited'] is True
        assert not report.get('allEpochRemainingPids', []) and report['privateVarsRemaining'] == 0
        rpo = report['actualRpoDifference']
        assert rpo['acknowledgedPostSnapshotWrites'] >= 1
        assert rpo['notRecoveredAcknowledgedWrites'] >= 1 and rpo['zeroLossClaimed'] is False
