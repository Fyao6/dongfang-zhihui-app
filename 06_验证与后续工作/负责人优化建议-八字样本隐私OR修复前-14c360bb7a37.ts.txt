   }catch{code=code||'object_identity_changed';}
   if(code){const fresh=await this.a.one('SELECT * FROM privacy_object_deletions WHERE id=?',r.id);if(fresh.lease_token===token){const terminal=code==='object_identity_changed'||fresh.attempts>=5;await this.a.run("UPDATE privacy_object_deletions SET status=?,lease_token='',lease_until=0,next_attempt=?,last_error=?,updated=?    current=await this.a.one('SELECT * FROM privacy_object_deletions WHERE id=?',r.id);items.push(this.object(current));
  }
  return {items,processed:items.length};
 }

export async function initPrivacyDomains(a:any){
 const schemaFingerprint=await catalogFingerprint(a),installed=await a.setting('privacy_domains_schema_v1');if(installed?.schemaFingerprint===schemaFingerprint)return;
 const service=new PrivacyDomains(a,()=>{throw Error('privacy_domains_init');}),sql=[...DDL];
 for(const [table,columns] of [['privacy_domain_jobs','id,owner,actor,request_id,mode,domains,as_of,config_data,pref_revision,closure_receipt_id,created'],['privacy_domain_targets','job_id,owner,domain,resource_id,write_at,version,nonce'],['privacy_object_deletions','id,owner,receipt_id,media_id,object_key,created'],['privacy_domain_receipts','id,owner,job_id,actor,request_id,fingerprint,created']])sql.push(`CREATE TRIGGER IF NOT EXISTS ${table}_identity BEFORE UPDATE OF ${columns} ON ${table} BEGIN SELECT RAISE(ABORT,'privacy_domain_immutable'); END`);
 sql.push("CREATE TRIGGER IF NOT EXISTS privacy_domain_confirmation_identity BEFORE UPDATE OF confirmed_by,confirmed_at ON privacy_domain_jobs WHEN OLD.confirmed_at>0 AND (NEW.confirmed_by IS NOT OLD.confirmed_by OR NEW.confirmed_at IS NOT OLD.confirmed_at) BEGIN SELECT RAISE(ABORT,'privacy_domain_confirmation_immutable'); END");
 for(const table of ['privacy_domain_jobs','privacy_domain_targets','privacy_media_tombstones','privacy_object_deletions'])sql.push(`CREATE TRIGGER IF NOT EXISTS ${table}_no_delete BEFORE DELETE ON ${table} BEGIN SELECT RAISE(ABORT,'privacy_domain_permanent'); END`);
 sql.push("CREATE TRIGGER IF NOT EXISTS privacy_media_tombstone_identity BEFORE UPDATE ON privacy_media_tombstones BEGIN SELECT RAISE(ABORT,'privacy_media_tombstone_immutable'); END");
 for(const [domain,s] of Object.entries(SOURCES)){
  for(const action of ['INSERT','UPDATE'])sql.push(`CREATE TRIGGER IF NOT EXISTS privacy_domain_track_${s.table}_${action.toLowerCase()} AFTER ${action} ON ${s.table} BEGIN INSERT INTO privacy_retention_tracking VALUES(NEW.${s.owner},'${domain}',NEW.${s.key},'',${NOW},1,lower(hex(randomblob(16))),'observed-write') ON CONFLICT(owner,scope,k1,k2) DO UPDATE SET write_at=excluded.write_at,version=privacy_retention_tracking.version+1,nonce=excluded.nonce,basis=excluded.basis; END`);
  sql.push(`CREATE TRIGGER IF NOT EXISTS privacy_domain_track_${s.table}_delete AFTER DELETE ON ${s.table} BEGIN DELETE FROM privacy_retention_tracking WHERE owner=OLD.${s.owner} AND scope='${domain}' AND k1=OLD.${s.key} AND k2=''; END`);
 }
 for(const [table,column] of await service.fields())for(const action of ['INSERT','UPDATE']){
  const condition=fieldReference(table,column,'NEW.'+quote(column),'dead.media_id');sql.push(`CREATE TRIGGER IF NOT EXISTS ${quote('privacy_deleted_'+table+'_'+column+'_'+action.toLowerCase())} BEFORE ${action}${action==='UPDATE'?' OF '+quote(column):''} ON ${quote(table)} BEGIN SELECT CASE WHEN EXISTS(SELECT 1 FROM privacy_media_tombstones dead WHERE ${condition}) THEN RAISE(ABORT,'privacy_deleted_media_reference') END; END`);
 }
 for(const action of ['INSERT','UPDATE'])sql.push(`CREATE TRIGGER IF NOT EXISTS privacy_deleted_media_${action.toLowerCase()} BEFORE ${action} ON media BEGIN SELECT CASE WHEN EXISTS(SELECT 1 FROM privacy_media_tombstones WHERE media_id=NEW.id) OR EXISTS(SELECT 1 FROM privacy_object_deletions WHERE object_key=NEW.object_key) THEN RAISE(ABORT,'privacy_deleted_media_identity') END; END`);
 await a.db.batch(sql.map(s=>a.stmt(s)).concat([a.stmt('INSERT OR IGNORE INTO privacy_domain_config VALUES(1,?)',J(DEFAULT))]));
 await a.run("INSERT INTO settings VALUES('privacy_domains_schema_v1',?) ON CONFLICT(key) DO UPDATE SET data=excluded.data",J({schemaFingerprint:await catalogFingerprint(a)}));
}
